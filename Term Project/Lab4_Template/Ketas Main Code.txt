;; setting up gpio ports B and C


BL System_Clock_Init        ; init the system clock

LDR r0, =RCC_BASE				; starts at the address of the RCC_BASE MODULE
	LDR r1, [r0, #RCC_AHB2ENR]		; offsets to the address of the AHB2ENR reg which controlls the gpio clocks
	ORR r1, r1, #0x00000006			;mask to enable gpio port B and C
	STR r1, [r0, #RCC_AHB2ENR]		; sends the bit mask back to the gpio memory
	
	
	
; GPIO B input

	LDR r0, =GPIOB_BASE
	LDR r1, [r0, #GPIO_MODER]
	AND r1, 0x0;			sets pins 1 2 3 to input
	STR r1, [r0, #GPIO_MODER]
	
	
; input type B

	LDR r0, =GPIOB_BASE
	LDR r1, [r0, #GPIO_PUPDR]
	AND r1, #0x0; 					set pins 1 2 3 and NO pull no pull down bc it is done externtally
	STR r1, [r0, #GPIO_PUPDR]
	
; GIPO C PINS output

	LDR r0, =GPIOC_BASE
	LDR r1, [r0, #GPIO_MODER]
	ORR r1, #0x55;					sets pins 0123 to output	
	STR r1, [r0, #GPIO_MODER]
	
; output type

	LDR r0, =GPIOC_BASE
	LDR r1, [r0, #GPIO_OTYPER]
	AND r1, #0x0;				rst
	ORR r1, r1, #0xF; 				sets pins 0123 to open drain					
	STR r1, [r0, #GPIO_OTYPER]
; End of GPIO Setup --------------------------------------------------------------------------------------------------------

loop ; Beginning of constant loop
	; Must set up the delay loop
	AND r10, r10, #0x00000000 ; Setting all of the bits to 0
	LDR r4, =0x00A00000 ; Loading delay onto core register #4
	
	; Pull all rows low
	LDR r0, =GPIOC_BASE
	LDR r1, [r0, #GPIO_ODR] 
	AND r1, r1, #0xF ; Set all row pins to low
	STR r1, [r0, #GPIO_ODR]
	
	; Invoking delay function
	BL delay
	
	; Scan columns to find if button is pressed - r0 has result
	BL checkCols
	MOV r5, r0 ; Move the result of the function into r5
	
	; Cycling through rows to determine the letter being pressed
	MOV r6, #0x1 ; Moving hex pin tester to r6 for loop control and masking
rowLoop
	CMP r6, #0x10
	BEQ afterLoop ; Ensures end of for loop
	BL checkRow
	CMP r0, #1 ; checkRow will be 1 if true
	BEQ positiveTest
	LSL r6, r6, #1 ; LSL the mask and counter over by 1
	BNE rowLoop
positiveTest
	EOR r6, r6, #0xF ; Invert values for active low
	AND r6, r6, #0xF ; Mask everything but bits 0-3
afterLoop
	; NOW COL IS IN R5 ROW IS IN R6 WITH BOTH ACTIVE LOW
	
	
displaykey
	STR	r5, [r8]
	LDR	r0, =char1
	;LDR r0, =str   ; First argument
	MOV r1, #1    ; Second argument
	BL USART2_Write
 	
	ENDP		

			
		

delay	PROC
	; Delay for software debouncing
	LDR	r2, =0x9999
delayloop
	SUBS	r2, #1
	BNE	delayloop
	BX LR
	
	ENDP



; RETURNS Pin 1-3 as HIGH depending on voltage
checkCols PROC
	; Checking columns of the keypad
	LDR r1, =GPIOB_BASE
	LDR r2, [r1, #GPIO_IDR]
	AND r2, r2, #0xE ; Mask everything but bits 1-3
	MOV r0, r2
	BX LR 
	
	ENDP


; RETURNS 1 if the row test is positive and 0 if it is negative
checkRow PROC
	; ROW 0 SET
	LDR r1, =GPIOC_BASE
	LDR r2, [r1, #GPIO_ODR] 
	AND r2, r2, #0xF ; Set all row pins to low
	ORR r2, r2, r6 ; Testing pin based on r6 - NOT ALTERING R6, JUST READING
	STR r2, [r1, #GPIO_ODR]
	
	; DELAY 1
	PUSH{LR}
	BL delay
	POP{LR}
	
	; TEST COL AGAIN
	PUSH{LR}
	BL checkCols
	POP{LR}
	MOV r3, r0
	CMP r3, #0xE ; Check if all cols are high with the new input
	MOVEQ r0, #1 ; Return 1 for true
	MOVNE r0 #0 ; Return 0 for false
	
	BX LR
	
	
	ENDP
